function frame(s:ActorState,time:number,distance=.02):RenderActorFrame{return {state:s,time,grounded:s.grounded,position:{...s.position},velocity:{...s.velocity},yaw:0,climb:s.climb,strike:s.strike,climbProgress:0,distance,angularDistance:0,reset:false};}

function fixture(layer=true){
 const nodes=graph.nodes.map((n:any)=>{const o=sourceManifest.jointNames.includes(n.name)?new Bone():new Group();o.name=n.name??'';if(n.translation)o.position.fromArray(n.translation);if(n.rotation)o.quaternion.fromArray(n.rotation);if(n.scale)o.scale.fromArray(n.scale);if(n.matrix){o.matrix.fromArray(n.matrix);o.matrix.decompose(o.position,o.quaternion,o.scale);}return o;});
 graph.nodes.forEach((n:any,i:number)=>(n.children??[]).forEach((child:number)=>nodes[i].add(nodes[child])));const model=new Group();for(const i of graph.scenes[graph.scene??0].nodes)model.add(nodes[i]);
 const root=new Group(),body=new Group();root.add(body);body.add(model);const owner=new Identity57Runtime005(model,original.clips,'test',sourceManifest,layer?recoil.clips:[],layer?recoilManifest:undefined);
 return {model,root,body,owner,head:model.getObjectByName('head') as Bone,foot:model.getObjectByName('FL_lower') as Bone};
}
const active=(f:ReturnType<typeof fixture>)=>f.owner.diagnostic.activeClips.filter(a=>a.name.endsWith('Moving048'));
function comparePreserved(a:ReturnType<typeof fixture>,b:ReturnType<typeof fixture>,includeHead=false){for(const name of sourceManifest.jointNames){if(name==='head'&&!includeHead)continue;const x=a.model.getObjectByName(name)!,y=b.model.getObjectByName(name)!;expect(x.position.toArray()).toEqual(y.position.toArray());expect(x.quaternion.toArray()).toEqual(y.quaternion.toArray());expect(x.scale.toArray()).toEqual(y.scale.toArray());expect(x.matrixWorld.elements).toEqual(y.matrixWorld.elements);}expect(a.root.matrixWorld.elements).toEqual(b.root.matrixWorld.elements);}
